type EndToEndTrace = {

### 边界处理

```ts
const gates = [
  verifyFrameBoundary, verifyIoGeneration, verifyAuthentication,
  verifyIdempotentCommit, verifyLobbyOwnership, verifyGmAudit,
  verifyReleaseRollback,
];

for (const gate of gates) await gate(trace);
```
