## 最小可运行实验

```javascript
function createCounter(start = 0) {
  let value = start;
  return {
    next(step = 1) {
      value += step;
      return value;
    },
    read() {
      return value;
    },
  };
}
const counter = createCounter(2);
console.log(counter.next(), counter.read());
```
