## 直觉引入

Lua 是嵌入式语言——它被设计为嵌入 C 程序中运行。但 C 和 Lua 有完全不同的类型系统：C 有 int/float/char*，Lua 有 8 种基本类型。它们怎么交换数据？

答案是**虚拟栈**。C 不能直接读写 Lua 变量——所有数据交换都通过一个栈进行。C 把数据压入栈（`lua_pushnumber`），Lua 从栈上取值；Lua 把返回值压入栈，C 从栈上取回。栈是两个世界之间的唯一桥梁。这个设计看似简单，但它是 Lua 嵌入式定位的核心——它让 C 和 Lua 完全解耦。

## 概念讲解

### 虚拟栈

<Term def="虚拟栈是 C 与 Lua 交换数据的唯一通道。栈底索引为 1，栈顶为 -1（或 lua_gettop(L)）。C 通过 lua_push* 压栈，lua_to* 取值，lua_pop 弹出。">

```c
// 栈的结构
// 索引:  1    2    3     (正索引，从底到顶)
//       "hi"  42   true
// 索引: -3   -2   -1     (负索引，从顶到底)

lua_gettop(L)      // 3（栈中元素个数 = 栈顶正索引）
lua_tostring(L, 1) // "hi"（正索引）
lua_tostring(L, -3)// "hi"（负索引，等价）
lua_tonumber(L, 2) // 42
lua_toboolean(L, -1)// true（栈顶）
lua_pop(L, 1)      // 弹出 1 个元素，现在栈顶是 42
```

</Term>

### 压栈与取值

```c
// 压栈：C 值 → Lua 栈
lua_pushnil(L);
lua_pushboolean(L, 1);        // true
lua_pushnumber(L, 3.14);
lua_pushinteger(L, 42);       // Lua 5.3+ 整数
lua_pushstring(L, "hello");
lua_pushlstring(L, buf, len); // 带长度的字符串

// 取值：Lua 栈 → C 值
int b = lua_toboolean(L, idx);
lua_Number n = lua_tonumber(L, idx);
lua_Integer i = lua_tointeger(L, idx);
const char *s = lua_tostring(L, idx);
size_t len;
const char *s2 = lua_tolstring(L, idx, &len);

// 类型检查
if (lua_isstring(L, idx)) { ... }
if (lua_type(L, idx) == LUA_TNUMBER) { ... }
```

### 编写 C 函数

```c
// 所有可被 Lua 调用的 C 函数签名：int (*)(lua_State *L)
// 返回值 = 压入栈的返回值个数

static int l_add(lua_State *L) {
    // 从栈底取参数（Lua 调用时自动压入）
    double a = lua_tonumber(L, 1);  // 第 1 个参数
    double b = lua_tonumber(L, 2);  // 第 2 个参数

    // 结果压栈
    lua_pushnumber(L, a + b);

    return 1;  // 返回 1 个值
}

static int l_sum(lua_State *L) {
    int n = lua_gettop(L);  // 获取参数个数
    double sum = 0;
    for (int i = 1; i <= n; i++) {
        sum += lua_tonumber(L, i);
    }
    lua_pushnumber(L, sum);
    return 1;
}
```

### 注册 C 函数

```c
// 方式1：注册到全局
lua_register(L, "add", l_add);  // Lua 中 add(1, 2) 可用

// 方式2：注册到 table（模块）
luaL_Reg mylib[] = {
    {"add", l_add},
    {"sum", l_sum},
    {NULL, NULL}  // 哨兵
};
luaL_newlib(L, mylib);  // 创建 table 并放入函数
lua_setglobal(L, "mylib");  // 命名为 mylib
// Lua 中 mylib.add(1, 2) 可用
```

### Lua 中调用注册的 C 函数

```lua
-- 注册后直接调用
print(add(1, 2))        -- 3
print(mylib.sum(1, 2, 3, 4, 5))  -- 15
```

### 完整的嵌入示例

```c
#include <lua.h>
#include <lauxlib.h>
#include <lualib.h>

int main() {
    // 1. 创建 Lua 状态
    lua_State *L = luaL_newstate();
    luaL_openlibs(L);  // 打开标准库

    // 2. 注册 C 函数
    lua_register(L, "add", l_add);

    // 3. 加载并执行 Lua 脚本
    if (luaL_dofile(L, "script.lua")) {
        fprintf(stderr, "Error: %s\n", lua_tostring(L, -1));
    }

    // 4. 调用 Lua 函数
    lua_getglobal(L, "greet");      // 压入 greet 函数
    lua_pushstring(L, "World");      // 压入参数
    if (lua_pcall(L, 1, 1, 0)) {    // 调用：1参数, 1返回值
        fprintf(stderr, "Error: %s\n", lua_tostring(L, -1));
    } else {
        const char *result = lua_tostring(L, -1);
        printf("%s\n", result);
        lua_pop(L, 1);  // 弹出返回值
    }

    // 5. 关闭
    lua_close(L);
    return 0;
}
```

<Step title="第一步：理解虚拟栈">

虚拟栈是 C 与 Lua 交换数据的唯一通道。C 不能直接操作 Lua 变量——必须通过栈。正索引从栈底 1 开始，负索引从栈顶 -1 开始。lua_gettop 返回栈中元素个数（= 栈顶正索引）。lua_pop 弹出指定数量的元素。

</Step>

<Step title="第二步：编写 C 函数">

所有可被 Lua 调用的 C 函数签名是 `int fn(lua_State *L)`。从栈上取参数（Lua 调用时自动压入），处理后将结果压栈，返回压入的返回值个数。lua_tonumber(L, 1) 取第一个参数，lua_pushnumber(L, result) 压入结果，return 1 表示返回 1 个值。

</Step>

<Step title="第三步：注册与嵌入">

lua_register(L, "name", fn) 注册 C 函数为 Lua 全局函数。luaL_newlib 创建包含多个函数的模块 table。嵌入流程：luaL_newstate 创建状态 → luaL_openlibs 打开标准库 → 注册 C 函数 → luaL_dofile 执行脚本 → lua_getglobal/pcall 调用 Lua 函数 → lua_close 关闭。

</Step>

</Stepper>

## 代码对照

```c
/* C 端：注册数学扩展库 */

#include <lua.h>
#include <lauxlib.h>
#include <math.h>

/* 计算平均值 */
static int l_average(lua_State *L) {
    int n = lua_gettop(L);       /* 参数个数 */
    double sum = 0;

    for (int i = 1; i <= n; i++) {
        /* 检查类型并取值 */
        luaL_checktype(L, i, LUA_TNUMBER);
        sum += lua_tonumber(L, i);
    }

    lua_pushnumber(L, sum / n);  /* 压入结果1：平均值 */
    lua_pushnumber(L, sum);      /* 压入结果2：总和 */
    return 2;                     /* 返回2个值 */
}

/* 字符串分割 */
static int l_split(lua_State *L) {
    const char *s = luaL_checkstring(L, 1);
    const char *sep = luaL_checkstring(L, 2);

    lua_newtable(L);  /* 创建结果 table */
    int i = 1;

    const char *p = s;
    while (*p) {
        const char *found = strstr(p, sep);
        if (!found) {
            lua_pushstring(L, p);
            lua_rawseti(L, -2, i++);
            break;
        }
        lua_pushlstring(L, p, found - p);
        lua_rawseti(L, -2, i++);
        p = found + strlen(sep);
    }

    return 1;  /* 返回 table */
}

/* 注册库 */
static const luaL_Reg mymathlib[] = {
    {"average", l_average},
    {"split",   l_split},
    {NULL, NULL}
};

int luaopen_mymath(lua_State *L) {
    luaL_newlib(L, mymathlib);
    return 1;
}
```

```lua
-- Lua 端：使用注册的 C 函数
local mymath = require("mymath")

local avg, total = mymath.average(10, 20, 30, 40, 50)
print(string.format("平均=%.1f, 总和=%.1f", avg, total))
-- 平均=30.0, 总和=150.0

local parts = mymath.split("a,b,c,d", ",")
for i, v in ipairs(parts) do
    print(i, v)  -- 1 a / 2 b / 3 c / 4 d
end
```

## 常见误区

<Callout type="trap">

**误区：忘记弹出栈上的值**

每次压栈操作都会增加栈深度。如果 C 函数返回时不弹出中间值，栈会持续增长导致内存泄漏。规则：C 函数返回前，栈上应该只保留要返回的值。lua_pop(L, n) 弹出 n 个值。lua_setglobal/lua_rawset 等操作会自动消耗栈上的值。

</Callout>

<Callout type="trap">

**误区：不检查参数类型导致崩溃**

`lua_tonumber(L, 1)` 在参数不是数字时返回 0 而非报错——这可能导致静默 bug。用 `luaL_checknumber(L, 1)` 替代——它在类型不匹配时抛出 Lua 错误（可通过 pcall 捕获），而非让 C 程序崩溃。所有外部输入都应该用 check 系列函数验证类型。

</Callout>

<Exercises>

**问题 1**：解释 Lua 虚拟栈的索引规则。正索引和负索引有什么区别？

<Answer>

Lua 虚拟栈有两种索引方式：

**正索引**：从栈底到栈顶，1, 2, 3, ..., n。1 是栈底（最早压入的），n 是栈顶（最后压入的）。n = lua_gettop(L)。

**负索引**：从栈顶到栈底，-1, -2, -3, ..., -n。-1 是栈顶（最后压入的），-n 是栈底。

```c
// 假设栈中有 3 个元素（从底到顶）："hi", 42, true
lua_tostring(L, 1)   == lua_tostring(L, -3)  // "hi"
lua_tonumber(L, 2)   == lua_tonumber(L, -2)  // 42
lua_toboolean(L, 3)  == lua_toboolean(L, -1) // true
```

负索引的好处：不需要知道栈中有多少元素就能访问栈顶。`lua_tostring(L, -1)` 总是取栈顶元素，无论栈深度如何。这在处理返回值和错误信息时非常方便。

</Answer>

**问题 2**：编写一个 C 函数 `l_repeat_str`，接受字符串和数字 n，返回字符串重复 n 次的结果。如何在 Lua 中注册并调用？

<Answer>

```c
#include <string.h>
#include <lauxlib.h>

static int l_repeat_str(lua_State *L) {
    /* 1. 从栈上取参数 */
    const char *s = luaL_checkstring(L, 1);  /* 第1参数：字符串 */
    int n = luaL_checkinteger(L, 2);          /* 第2参数：次数 */

    if (n < 0) {
        luaL_error(L, "repeat count must be non-negative");
    }

    /* 2. 计算结果长度 */
    size_t len = strlen(s);
    size_t total = len * n;

    /* 3. 创建结果字符串并压栈 */
    luaL_Buffer b;
    luaL_buffinit(L, &b);
    for (int i = 0; i < n; i++) {
        luaL_addlstring(&b, s, len);
    }
    luaL_pushresult(&b);

    /* 4. 返回 1 个值 */
    return 1;
}

/* 注册 */
void register_funcs(lua_State *L) {
    lua_register(L, "repeat_str", l_repeat_str);
}
```

Lua 中调用：
```lua
-- 注册后直接调用
print(repeat_str("ab", 3))  -- ababab
print(repeat_str("-", 10))  -- ----------
```

关键点：
1. C 函数签名 `int fn(lua_State *L)`
2. luaL_checkstring/checkinteger 取参数并检查类型
3. luaL_Buffer 处理可变长度字符串拼接
4. return 1 表示返回 1 个值
5. lua_register 注册为全局函数

</Answer>

<GlossaryItem term="虚拟栈">

C 与 Lua 交换数据的唯一通道。C 通过 lua_push* 压栈，lua_to* 取值。正索引从栈底 1 开始，负索引从栈顶 -1 开始。lua_gettop 返回元素个数，lua_pop 弹出元素。

</GlossaryItem>

<GlossaryItem term="C 函数签名">

所有可被 Lua 调用的 C 函数签名：`int fn(lua_State *L)`。从栈上取参数，结果压栈，返回值个数。luaL_check* 系列函数取参数并验证类型。

</GlossaryItem>

<GlossaryItem term="lua_register">

注册 C 函数为 Lua 可调用函数。lua_register(L, "name", fn) 注册到全局。luaL_newlib 创建包含多个函数的模块 table。注册后 Lua 中直接 name(args) 调用。

</GlossaryItem>

<GlossaryItem term="嵌入式定位">

Lua 是嵌入式语言——通过 C API 嵌入宿主程序。C 注册函数供 Lua 调用，Lua 调用 C 函数通过栈交换数据。这让宿主程序获得脚本化能力——用 Lua 编写配置/扩展逻辑，核心功能用 C 实现。

</GlossaryItem>
