## 为什么从可验证状态开始

本页按官方第二版在线目录重建 **Chapter 14. The Ethereum Virtual Machine**，覆盖该正式单元的 52 个二至五级标题。教学顺序保留原目录的依赖关系，但正文全部独立撰写：每个标题都转成对象、前状态、输入、规则、后状态和失败证据，避免只给术语摘要。

EVM是确定性的256位栈机，代码、栈、内存、calldata、returndata、持久存储和执行上下文共同决定一帧调用。操作码Gas与状态访问规则限制资源。 阅读时把“客户端或网页显示了什么”和“协议能验证什么”分开：钱包、浏览器、RPC与索引器是观察面，规范编码、签名、交易、收据、区块、状态根、证明和最终化检查点才是可复核对象。任何结论都要绑定链ID、客户端版本、区块坐标与原始输入。

实验统一在本地开发链或公开测试网完成，私钥只使用一次性测试向量。先写预期，再运行；先保存原始对象，再看格式化界面。正常样本证明主路径，边界样本选择零值、最大值、恰好阈值或时序边缘，单故障样本只改变链ID、nonce、一个字节、一个签名、一个价格或一个证明条件。

每一步执行由当前PC、栈、内存、存储和剩余Gas唯一决定；异常终止回滚当前调用的状态变化但保留规定的Gas消耗。 若执行失败，证据必须说明首个拒绝规则、错误数据、消耗的资源以及状态是否回滚；若执行成功，还要区分交易池接纳、区块包含、执行状态与PoS最终性，不能把交易哈希当成最终结算。

通过标准是：能逐步执行一段短字节码，记录栈和内存，计算关键Gas，并比较CALL、DELEGATECALL与STATICCALL上下文。 独立复核者应能从空目录和相同测试向量重建结果；缺失来源、版本、输入、前状态、后状态或失败证据任一项，结论都不能进入下一页。

```typescript
type EthereumEvidence = {
  chainId: number;
  block: string;
  beforeRoot: string;
  inputHash: string;
  afterRoot?: string;
  error?: string;
};

export function evidenceGate(run: EthereumEvidence) {
  return {
    page: "met2-14-ethereum-virtual-machine",
    accepted: Boolean(run.afterRoot || run.error),
    immutableInput: run.inputHash,
  };
}
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.26;

contract InvariantProbe {
    uint256 public value;
    event Applied(uint256 beforeValue, uint256 afterValue);

    function apply(uint256 next) external {
        uint256 beforeValue = value;
        require(next >= beforeValue, "non-monotonic");
        value = next;
        emit Applied(beforeValue, next);
    }
}
```

```typescript
function applyCandidate(
  state: bigint,
  next: bigint,
  gasUsed: bigint,
  gasLimit: bigint,
) {
  if (gasUsed > gasLimit)
    return { accepted: false, state, reason: "out-of-gas" };
  if (next < state)
    return { accepted: false, state, reason: "invalid-transition" };
  return { accepted: true, state: next, reason: null };
}
```

第一段把成功与失败放进同一证据合同；第二段给出可观察事件和显式不变量；第三段证明拒绝路径保持原状态。真实系统应使用成熟客户端、ABI、密码与证明库，并补齐并发、重组、超时、持久化、隐私和密钥隔离。
