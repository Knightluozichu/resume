最小实现只暴露本章关键状态，不把因果藏进通用框架。

```csharp
public readonly record struct PathRequest(int Start, int Goal, int MapVersion);

static float Priority(float cost, float heuristic) => cost + heuristic;

// 开放集按 Priority 排序；只有得到更低 cost 时才更新父节点。
```

统一证据契约把结构、版本和故障计划放在同一结果中。

```ts
type LeadProgramEvidence = {
  chapter: "u3ap-10-map-pathfinding";
  runtimeVersion: string;
  sample: "baseline" | "boundary" | "failure";
  stages: readonly string[];
  firstDeviation: string | null;
  decision: "pass" | "rejected";
};

function canRelease(run: LeadProgramEvidence): boolean {
  return run.stages.length === 6 &&
    (run.sample !== "failure" || run.decision === "rejected");
}
```
